The Syntax

/* Base token */
--brand: oklch(63% 0.18 264);

/* Lighter hover state */
background: oklch(from var(--brand) calc(l + 0.10) c h);

/* Darker active state */
background: oklch(from var(--brand) calc(l - 0.12) c h);

/* Translucent shadow */
box-shadow: oklch(from var(--brand) l c h / 30%);

/* Completely different hue, same feel */
background: oklch(from var(--brand) l c 340);
