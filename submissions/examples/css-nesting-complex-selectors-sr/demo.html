.feature-card {
  transition:
    transform 220ms ease,
    box-shadow 220ms ease;

  &:hover {
    transform: translateY(-6px);

    .feature-card__link {
      gap: 0.75rem;
    }
  }

  &:focus-within {
    outline: 2px solid currentColor;
  }

  &:has(.feature-card__button:focus-visible) {
    box-shadow:
      0 0 0 4px rgb(99 102 241 / 18%);
  }
}
