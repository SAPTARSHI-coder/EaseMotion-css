// easemotion/engine/parser.js Token Loop
for (const token of tokens) {
  // 1. Animation names take precedence over easing map
  if (ANIMATION_NAMES.has(token)) {
    ast.animation = token; // 'bounce' captures here
    continue;
  }
  // 2. Canonical easing aliases (use 'spring' for cubic-bezier)
  if (EASING_MAP[token]) {
    ast.easing = EASING_MAP[token];
    continue;
  }
