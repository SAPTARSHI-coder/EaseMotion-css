Core CSS Formula (style.css)

transform: translate(
  calc(cos(var(--angle)) * var(--radius)),
  calc(sin(var(--angle)) * var(--radius))
);

/* Each item receives its angle via inline style: */
<div class="orbit-item" style="--angle: 60deg;">…</div>
--angle: 0deg --angle: 60deg --angle: 120deg --angle: 180deg --angle: 240deg --angle: 300deg
