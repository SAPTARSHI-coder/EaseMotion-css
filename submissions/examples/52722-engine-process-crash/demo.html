Engine Runtime Crash Fix

Demonstrates the fix for #52722 where the Motion Engine crashes with ReferenceError: process is not defined when parsing invalid em="" attributes in browser environments.

Before Fix (Broken)

// easemotion/engine/runtime.js — ORIGINAL (line 55)
function processElement(el) {
  const ast = parse(value);
  if (!ast) {
    // CRASHES in browser — `process` is undefined!
    if (process?.env?.NODE_ENV !== 'production') {
      console.warn(`Unknown animation name: ${value}`);
    }
    return;
  }
}
❌ ReferenceError: process is not defined

After Fix (Working)

// easemotion/engine/runtime.js — FIXED (line 55)
function processElement(el) {
  const ast = parse(value);
  if (!ast) {
    // SAFE — typeof check prevents ReferenceError
    if (typeof process === 'undefined' || process.env?.NODE_ENV !== 'production') {
      console.warn(`Unknown animation name: ${value}`);
    }
    return;
  }
}
✅ Graceful warning — no crash
