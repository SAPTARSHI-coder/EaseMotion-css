// Fetch workspace metrics from EaseMotion API
async function fetchMetrics(apiKey) {
  const response = await fetch('https://api.easemotion.dev/v1/quota', {
    headers: { 'Authorization': `Bearer ${apiKey}` }
  });
  if (!response.ok) throw new Error('API Request Failed');
  return await response.json();
}

import requests
def fetch_metrics(api_key: str) -> dict:
    headers = {"Authorization": f"Bearer {api_key}"}
    response = requests.get("https://api.easemotion.dev/v1/quota", headers=headers)
    response.raise_for_status()
    return response.json()

// Fetch metrics using reqwest in Rust
pub async fn fetch_quota(key: &str) -> Result<Quota, reqwest::Error> {
    let client = reqwest::Client::new();
    let res = client.get("https://api.easemotion.dev/v1/quota")
        .header("Authorization", format!("Bearer {}", key))
        .send().await?.json::<Quota>().await?;
    Ok(res)
}
