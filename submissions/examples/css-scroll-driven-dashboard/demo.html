How It Works

@property + scroll counter 
@property --num {
  syntax: '<integer>';
  initial-value: 0;
  inherits: false;
}

.stat-value::after {
  counter-reset: num var(--num);
  content: counter(num);
  animation: count-up 2s ease-out forwards;
  animation-timeline: view();
  animation-range: entry 0% entry 100%;
}

@keyframes count-up {
  to { --num: var(--to); }
}
@property + view-timeline skill bars 
@property --pct {
  syntax: '<percentage>';
  initial-value: 0%;
  inherits: false;
}

.skill-fill {
  width: var(--pct);
  animation: fill-bar 2s ease-out forwards;
  animation-timeline: view();
}

@keyframes fill-bar {
  to { --pct: calc(var(--skill) * 1%); }
}
scroll() progress bar 
.scroll-progress {
  scroll-timeline-name: --page;
  scroll-timeline-axis: block;
  animation: page-progress linear;
  animation-timeline: --page;
  transform-origin: left;
}

@keyframes page-progress {
  from { transform: scaleX(0); }
  to { transform: scaleX(1); }
}
