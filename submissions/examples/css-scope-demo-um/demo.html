/* Elegant Glass Profile Card Scope */
@scope (.profile-card) {
  :scope {
    background: linear-gradient(135deg, rgba(99, 102, 241, 0.15), rgba(99, 102, 241, 0.05));
  }
  .avatar {
    border-radius: 50%;
    border: 3px solid #6366f1;
  }
  .title {
    color: #a5b4fc;
  }
}

/* Cyberpunk Card Scope */
@scope (.cyber-card) {
  :scope {
    border: 2px solid #ff007f;
  }
  .title {
    color: #ff007f;
    text-transform: uppercase;
  }
}

/* Scoping Limits Demo */
@scope (.limit-demo) to (.limit-boundary) {
  .title {
    color: #10b981;
    text-decoration: underline;
  }
}
