EaseMotion CSS

CSS Variable Dark Mode Toggle

All colors are driven by CSS custom properties. Flipping the toggle updates data-theme="dark" on the html element, switching every variable at once with a smooth transition.

Usage

<html data-theme="light">
...
<link rel="stylesheet" href="style.css"/>
<button class="ease-theme-toggle" id="themeToggle">
  <span class="ease-theme-toggle__track">
    <span class="ease-theme-toggle__thumb"></span>
  </span>
</button>
<script>
  const btn = document.getElementById('themeToggle');
  const html = document.documentElement;
  const saved = localStorage.getItem('ease-theme');
  if (saved) html.dataset.theme = saved;
  btn.addEventListener('click', () => {
    const next = html.dataset.theme === 'dark' ? 'light' : 'dark';
    html.dataset.theme = next;
    localStorage.setItem('ease-theme', next);
  });
</script>
