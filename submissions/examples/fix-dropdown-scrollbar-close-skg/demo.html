Fix 3 — JS-driven dropdowns: close on mousedown outside, not blur

✅ Guidance — correct event for JS close logic 
// ❌ WRONG — closes on blur, triggered by scrollbar click
toggle.addEventListener('blur', () => menu.hidden = true);

// ✅ CORRECT — close only on mousedown outside the container
document.addEventListener('mousedown', (e) => {
  if (!container.contains(e.target)) {
    menu.hidden = true;
  }
});

// Also close on Escape key for accessibility
document.addEventListener('keydown', (e) => {
  if (e.key === 'Escape') {
    menu.hidden = true;
    toggle.focus();
  }
});
