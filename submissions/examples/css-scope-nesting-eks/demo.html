CSS Nesting & in Action

Hover Effects

.demo-card--hover {
  &:hover {
    .demo-card__icon { scale: 1.2 }
  }
  &:focus-within {
    outline: 2px solid var(--accent)
  }
}

Nested Media

.demo-card--nested {
  display: grid;
  @media (width >= 600px) {
    grid-template-columns: 1fr 1fr
  }
}

Nested Layers

@scope (.demo-card--layer) {
  @layer base {
    & { padding: 1rem }
  }
  @layer theme {
    & { background: var(--surface) }
  }
}
