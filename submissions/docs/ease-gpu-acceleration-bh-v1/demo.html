Best Practices

/* Use transform for GPU acceleration */
.ease-fade-in {
  opacity: 0;
  transform: translateY(20px);
  transition: all 0.3s ease;
}
.ease-fade-in.visible {
  opacity: 1;
  transform: translateY(0);
}

/* Use will-change sparingly */
.animated-element {
  will-change: transform, opacity;
}

/* Avoid animating these (triggers layout) */
.bad { animation: width 0.3s; }      /* Layout */
.bad { animation: margin 0.3s; }    /* Layout */
.bad { animation: padding 0.3s; }    /* Layout */

/* Good (compositor-only) */
.good { animation: transform 0.3s; }  /* GPU */
.good { animation: opacity 0.3s; }     /* GPU */
.good { animation: filter 0.3s; }      /* GPU */
